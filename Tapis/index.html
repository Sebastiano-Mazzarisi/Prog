--------------------------------------------------------------------------- # Comandi (avvio, pausa, stop, imposta velocità) # # Il pacchetto di comando a 23 byte è lo stesso in entrambe le varianti di # --------------------------------------------------------------------------- VELOCITA_MIN = 1.0 VELOCITA_MAX = 12.0 # limite dichiarato dal produttore (1-12 km/h) VELOCITA_STEP = 0.5 _USER_ID_DEFAULT = 58965456623 def _checksum_comando(pacchetto: bytearray) -> int: c = 0 for b in pacchetto[1:21]: c ^= b return c def _crea_pacchetto_comando(velocita_raw: int, magical_i11: int, cmd_type: int, incline: int = 0) -> bytes: """cmd_type: 4 = Avvia/Imposta velocità, 2 = Pausa, 0 = Stop. magical_i11: 1 per avvio/pausa/stop, 5 per il solo cambio di velocità.""" p = bytearray(23) p[0] = 0x6A p[1] = 0x17 p[6] = (velocita_raw >> 8) & 0xFF p[7] = velocita_raw & 0xFF p[8] = magical_i11 p[9] = incline p[10] = 80 # peso di default (kg) p[11] = 0 p[12] = cmd_type & 0xF7 # modalità km/h for i in range(8): p[13 + i] = (_USER_ID_DEFAULT >> (56 - i * 8)) & 0xFF p[21] = _checksum_comando(p) p[22] = 0x43 return bytes(p) def _clamp_velocita(v: float) -> float: return max(VELOCITA_MIN, min(VELOCITA_MAX, v)) def pacchetto_avvia(velocita_kmh: float) -> bytes: return _crea_pacchetto_comando(round(_clamp_velocita(velocita_kmh) * 1000), 1, 4) def pacchetto_pausa() -> bytes: return _crea_pacchetto_comando(0, 1, 2) def pacchetto_stop() -> bytes: return _crea_pacchetto_comando(0, 1, 0) def pacchetto_imposta_velocita(velocita_kmh: float) -> bytes: return _crea_pacchetto_comando(round(_clamp_velocita(velocita_kmh) * 1000), 5, 4) class CodaComandi: """Ponte semplice e thread-safe tra il thread del server web (dove arrivano i click dei pulsanti) e il thread BLE (asyncio) che deve scrivere il pacchetto sulla caratteristica giusta.""" def __init__(self): self._lock = threading.Lock() self._prossimo: Optional[bytes] = None def imposta(self, pacchetto: bytes): with self._lock: self._prossimo = pacchetto def preleva(self) -> Optional[bytes]: with self._lock: p = self._prossimo self._prossimo = None return p COMANDI = CodaComandi() # --------------------------------------------------------------------------- # Parsing del pacchetto dati # --------------------------------------------------------------------------- def _checksum_ok(chunk: bytes) -> bool: """Verifica il checksum XOR usato dal firmware (byte penultimo del pacchetto, calcolato sui byte da 1 a lunghezza-3).""" length = len(chunk) - 2 if length < 3: return False checksum = chunk[1] for i in range(2, length): checksum ^= chunk[i] return checksum == chunk[-2] def _decode_fields(p: bytes) -> dict: speed = (p[3] << 8) | p[4] target_speed = (p[5] << 8) | p[6] distance = (p[7] << 24) | (p[8] << 16) | (p[9] << 8) | p[10] incline = p[11] heart_rate = p[13] steps = (p[14] << 24) | (p[15] << 16) | (p[16] << 8) | p[17] calories = (p[18] << 8) | p[19] duration_ms = (p[20] << 24) | (p[21] << 16) | (p[22] << 8) | p[23] firmware_version = p[25] flags = p[26] unit_mode = 1 if (flags & 128) == 128 else 0 state_bits = flags & 24 if state_bits == 24: running_state = 0 elif state_bits == 8: running_state = 1 elif state_bits == 16: running_state = 2 else: running_state = 3 duration_s = duration_ms / 1000 if firmware_version > 19 else float(duration_ms) return { "velocita": round(speed / 1000, 2), "velocita_target": round(target_speed / 1000, 2), "unita_velocita": "mph" if unit_mode else "km/h", "distanza": round(distance / 1000, 2), "unita_distanza": "mi" if unit_mode else "km", "inclinazione": incline, "frequenza_cardiaca": heart_rate or None, "passi": steps, "calorie": calories, "tempo_s": duration_s, "stato": STATI[running_state], "stato_idx": running_state, "firmware": firmware_version, } def parse_payload(raw: bytes) -> Optional[dict]: """Interpreta una notifica BLE grezza provando entrambi gli offset di header noti (0 o 4 byte) e convalidando il checksum. Se nessuno dei due è valido, restituisce comunque il tentativo più plausibile marcato come non verificato, utile in modalità debug.""" candidati = [] for offset in (4, 0): chunk = raw[offset:] if len(chunk) < 31: continue ok = _checksum_ok(chunk) try: campi = _decode_fields(chunk) except IndexError: continue candidati.append((ok, offset, campi)) if not candidati: return None for ok, offset, campi in candidati: if ok: campi["_offset"] = offset campi["_checksum_ok"] = True campi["_raw_hex"] = raw.hex() return campi ok, offset, campi = candidati[0] campi["_offset"] = offset campi["_checksum_ok"] = False campi["_raw_hex"] = raw.hex() return campi #
